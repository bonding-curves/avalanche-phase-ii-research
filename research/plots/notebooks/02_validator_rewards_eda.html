from types import SimpleNamespace
import numpy as np, pandas as pd, panel as pn, param as pm
from bokeh import models as bm, palettes as bp, plotting as bk, transform as bt

pn.extension()
POLICY = SimpleNamespace(origin=0, maximum_rate=.12, maximum_weeks=52, exponent=2, curve_points=128)
STYLE = SimpleNamespace(focus_width=4, focus_alpha=1, context_width=1.5, context_alpha=.22,
                        outcome_size=10, filled=1, hollow=0, plot_size=(900, 520), estimated_color="#9e9e9e")
LINE_DASH = {"2 days": "dotted", "2 weeks": "solid"}
SCENARIOS = pd.DataFrame([
    ("S1", "2 weeks", 2, .10, 10.0), ("F1", "2 weeks", 2, .09, np.nan),
    ("S3", "2 weeks", 2, .08, 16.9), ("F2", "2 weeks", 2, .07, np.nan),
    ("S2", "2 days", 2/7, .10, np.nan), ("S5", "2 days", 2/7, .09, 12.5),
    ("S4", "2 days", 2/7, .08, 16.7), ("S6", "2 days", 2/7, .07, 20.9),
], columns=["scenario", "minimum", "minimum_weeks", "r_min", "mean_weeks"])

def protocol_yield(duration, minimum_rate):
    share = duration / POLICY.maximum_weeks
    return minimum_rate * share + (POLICY.maximum_rate - minimum_rate) * share**POLICY.exponent

def estimate_means(scenarios):
    outcomes = scenarios.rename(columns={"mean_weeks": "reported_mean_weeks"}).copy()
    outcomes["mean_weeks"] = outcomes.reported_mean_weeks
    outcomes["mean_evidence"] = "reported"
    for minimum, group in outcomes.groupby("minimum"):
        known = group.dropna(subset=["reported_mean_weeks"])
        slope, intercept = np.polyfit(known.r_min, known.reported_mean_weeks, 1)
        missing = group.reported_mean_weeks.isna()
        indices = group.index[missing]
        outcomes.loc[indices, "mean_weeks"] = slope * group.loc[indices, "r_min"] + intercept
        inside = group.loc[indices, "r_min"].between(known.r_min.min(), known.r_min.max())
        outcomes.loc[indices, "mean_evidence"] = np.where(inside, "interpolated", "extrapolated")
    return outcomes.assign(yield_rate=lambda x: protocol_yield(x.mean_weeks, x.r_min))

def curve_source(scenarios):
    curves = scenarios.copy()
    curves["xs"] = curves.minimum_weeks.map(
        lambda start: np.linspace(start, POLICY.maximum_weeks, POLICY.curve_points))
    curves["ys"] = [protocol_yield(x, rate) for x, rate in zip(curves["xs"], curves.r_min)]
    curves["alpha"], curves["width"] = STYLE.context_alpha, STYLE.context_width
    curves["dash"] = curves.minimum.map(LINE_DASH)
    curves["legend_x"] = curves["legend_y"] = np.nan
    curves["legend_xs"] = curves["legend_ys"] = [[np.nan] * 2 for _ in range(len(curves))]
    curves["legend_mean"] = np.nan
    curves["legend_fill"] = np.where(curves.minimum.eq("2 weeks"), STYLE.filled, STYLE.hollow)
    return bm.ColumnDataSource(curves)

def focus_source(source, selection):
    minimums, rates = np.asarray(source.data["minimum"]), np.asarray(source.data["r_min"])
    selected = (minimums == selection[0]) & (rates == selection[1])
    lookup = {(minimum, rate): mean for minimum, rate, mean in zip(minimums, rates, source.data["mean_weeks"])}
    legend_means = [lookup[(minimum, selection[1])] for minimum in minimums]
    source.patch({
        "alpha": [(slice(len(selected)), np.where(selected, STYLE.focus_alpha, STYLE.context_alpha))],
        "width": [(slice(len(selected)), np.where(selected, STYLE.focus_width, STYLE.context_width))],
        "legend_mean": [(slice(len(selected)), legend_means)],
    })

def reward_figure(source, scenarios):
    means = scenarios.mean_weeks.dropna()
    colors = bm.LinearColorMapper(palette=bp.Viridis256, low=means.min(), high=means.max(), nan_color=STYLE.estimated_color)
    plot = bk.figure(width=STYLE.plot_size[0], height=STYLE.plot_size[1],
                     x_range=(POLICY.origin, POLICY.maximum_weeks), y_range=(POLICY.origin, POLICY.maximum_rate),
                     title="Avalanche reward curves and reported outcomes",
                     x_axis_label="Validation duration (weeks)",
                     y_axis_label="Normalized protocol yield Y(d) / kappa(t)")
    plot.multi_line("xs", "ys", source=source, line_color=bt.transform("mean_weeks", colors),
                    line_alpha="alpha", line_width="width", line_dash="dash")
    outcomes = estimate_means(scenarios).assign(
        color_mean=lambda x: x.mean_weeks.where(x.mean_evidence.eq("reported")),
        fill_alpha=lambda x: np.where(x.minimum.eq("2 weeks"), STYLE.filled, STYLE.hollow))
    outcome_source = bm.ColumnDataSource(outcomes)
    points = plot.scatter("mean_weeks", "yield_rate", source=outcome_source, size=STYLE.outcome_size,
                          fill_color=bt.transform("color_mean", colors), fill_alpha="fill_alpha",
                          line_color=bt.transform("color_mean", colors),
                          line_width=2)
    plot.add_tools(bm.HoverTool(renderers=[points], tooltips=[
        ("Scenario", "@scenario"), ("Minimum", "@minimum"), ("r_min", "@r_min{0.00}"),
        ("Mean", "@mean_weeks{0.00} weeks"), ("Evidence", "@mean_evidence"),
    ]))
    legends = []
    for minimum in LINE_DASH:
        index = list(source.data["minimum"]).index(minimum)
        view = bm.CDSView(filter=bm.IndexFilter(indices=[index]))
        line = plot.multi_line("legend_xs", "legend_ys", source=source, view=view,
                               line_color=bt.transform("legend_mean", colors),
                               line_dash="dash", line_width=STYLE.focus_width)
        point = plot.scatter("legend_x", "legend_y", source=source, view=view, size=STYLE.outcome_size,
                             fill_color=bt.transform("legend_mean", colors), fill_alpha="legend_fill",
                             line_color=bt.transform("legend_mean", colors), line_width=2)
        legends.append(bm.LegendItem(label=f"{minimum} mean duration", renderers=[line, point]))
    plot.add_layout(bm.Legend(items=legends, location="bottom_right"))
    plot.add_layout(bm.ColorBar(color_mapper=colors, title="Reported mean duration (weeks)"), "right")
    return plot, outcomes

class RewardCurveExplorer(pm.Parameterized):
    minimum_rate = pm.Selector(default=SCENARIOS.r_min.max(), objects=sorted(SCENARIOS.r_min.unique()))
    minimum_duration = pm.Selector(default="2 weeks", objects=list(LINE_DASH))
    def __init__(self, scenarios=SCENARIOS, **params):
        super().__init__(**params)
        self.source = curve_source(scenarios)
        figure, self.outcomes = reward_figure(self.source, scenarios)
        self.plot = pn.pane.Bokeh(figure)
        self.update_focus()
    @pm.depends("minimum_rate", "minimum_duration", watch=True)
    def update_focus(self):
        focus_source(self.source, (self.minimum_duration, self.minimum_rate))
        pn.io.push_notebook(self.plot)
    def view(self):
        controls = pn.Param(self, parameters=["minimum_rate", "minimum_duration"], widgets={"minimum_rate": pn.widgets.DiscreteSlider})
        return pn.Row(controls, self.plot)
